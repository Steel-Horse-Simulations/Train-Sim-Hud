BACKEND ROUTE CONVENTIONS (app.py)
================================================================================

List:
  @app.route("/api/YOUR_RESOURCE", methods=["GET"])
  def your_resource_list():
      items = your_db.list_items()
      return jsonify({"items": items})

Single:
  @app.route("/api/YOUR_RESOURCE/<int:item_id>", methods=["GET"])
  def your_resource_get(item_id):
      item = your_db.get_item(item_id)
      if not item: return jsonify({"error": "not_found"}), 404
      return jsonify(item)

PATCH:
  @app.route("/api/YOUR_RESOURCE/<int:item_id>", methods=["PATCH"])
  def your_resource_update(item_id):
      body = request.get_json(force=True, silent=True) or {}
      ok = your_db.update_item(item_id, body)
      if not ok: return jsonify({"error": "no_editable_fields_provided"}), 400
      return jsonify(your_db.get_item(item_id))

EDITABLE FIELDS PATTERN (db module)
================================================================================

EDITABLE_FIELDS = {"display_name", "notes", "is_visible"}

def update_item(item_id, fields):
    safe = {k: v for k, v in fields.items() if k in EDITABLE_FIELDS}
    if not safe: return False
    now = datetime.now().isoformat(timespec="seconds")
    conn = _connect()
    try:
        set_clause = ", ".join(f"{col} = ?" for col in safe) + ", updated_at = ?"
        conn.execute(f"UPDATE items SET {set_clause} WHERE id = ?",
                     list(safe.values()) + [now, item_id])
        conn.commit()
        return True
    finally:
        conn.close()
