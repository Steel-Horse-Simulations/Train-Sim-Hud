4. INLINE JS SYNTAX CHECK (every <script> block in every HTML page)
================================================================================
python3 -c "
import re, subprocess, os
fail = False
for f in sorted(os.listdir('pages')):
    if not f.endswith('.html'): continue
    with open(os.path.join('pages', f)) as fh: content = fh.read()
    for i, s in enumerate(re.findall(r'<script>(.*?)</script>', content, re.DOTALL)):
        tmp = f'/tmp/check_{f}_{i}.js'
        with open(tmp, 'w') as out: out.write(s)
        r = subprocess.run(['node', '--check', tmp], capture_output=True, text=True)
        if r.returncode != 0: fail = True
        print(f'{f} script {i}:', 'OK' if r.returncode==0 else 'FAIL: '+r.stderr.strip())
        os.remove(tmp)
print('ANY FAIL:', fail)
"
